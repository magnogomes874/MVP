WHERE codigo = ?", (nova, cod)) conn.commit() conn.close() display(HTML("✔️ Estoque atualizado com sucesso!")) atualizar_interface() btn_mov.on_click(registrar_movimentacao) # Aba 2: Novo Cadastro txt_cod = widgets.Text(description="Código SKU:") drop_marca = widgets.Dropdown(options=MARCAS, description="Marca:") drop_cat = widgets.Dropdown(options=CATEGORIAS, description="Categoria:") txt_desc = widgets.Text(description="Descrição:") num_qtd = widgets.FloatText(value=0.0, description="Qtd Inicial:") num_min = widgets.FloatText(value=3.0, description="Est. """, (txt_cod.value.strip().upper(), drop_marca.value, drop_cat.value, txt_desc.value.strip(), num_qtd.value, num_min.value, num_custo.value, num_venda.value)) conn.commit() conn.close() display(HTML("✔️ Produto cadastrado com sucesso!")) txt_cod.value = "" txt_desc.value = "" atualizar_interface() except sqlite3.IntegrityError: display(HTML("❌ Erro: Código SKU já existe.")) btn_cad.on_click(registrar_cadastro) # Aba 3: Editar / Excluir dropdown_edit = widgets.Dropdown(description="Produto:", layout=widgets.Layout(width='65%')) edit_marca = widgets.Dropdown(options=MARCAS, description="Marca:") edit_cat = widgets.Dropdown(options=CATEGORIAS, description="Categoria:") edit_desc = widgets.Text(description="Descrição:") edit_min = widgets.FloatText(description="Est.
